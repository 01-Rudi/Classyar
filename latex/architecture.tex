\section{نمای کلی معماری}

\subsection{سبک معماری}

\textbf{Monolith + Layered Architecture}

\begin{itemize}
\tightlist
\item
  یک برنامهٔ واحد
\item
  سه لایهٔ اصلی: Presentation، Business، Data
\item
  مناسب تیم کوچک (۵ نفر) و پروژهٔ ۱۴ هفته‌ای
\item
  ساده برای شروع، قابل گسترش
\end{itemize}

\subsection{دیاگرام لایه‌ها}

\begin{LTR}
\begin{verbatim}
+——————————————+
|  Presentation Layer                      |
|  React (SPA)                             |
+———————+——————–+
                      | HTTP / REST (JSON)
+———————v——————–+
|  Business Layer                          |
|  Node.js + Express                       |
+———————+——————–+
                      | Prisma (SQL)
+———————v——————–+
|  Data Layer                              |
|  PostgreSQL + File Storage               |
+——————————————+
\end{verbatim}
\end{LTR}

\subsection{جریان داده}

کاربر ← مرورگر (React) ← API (Express) ← پایگاه داده (PostgreSQL) و ذخیرهٔ فایل

درخواست از کاربر به Frontend، سپس به API و در پایان به پایگاه داده می‌رسد؛ پاسخ همین مسیر را برمی‌گردد.

\section{تصمیم‌های معماری (ADR)}

\subsection{ADR-01: چرا Monolith؟}

\textbf{تصمیم:} معماری Monolith

\textbf{دلیل:}
- تیم کوچک (۵ نفر) و زمان محدود (۱۴ هفته)
- پیچیدگی کم و یک زبان (JavaScript) برای کل تیم
- سرعت توسعه بالاتر
- استقرار ساده

\subsection{ADR-02: چرا React؟}

\textbf{تصمیم:} React برای Frontend

\textbf{دلیل:}
- محبوب و مستند
- اکوسیستم بزرگ
- SPA برای تجربهٔ کاربری بهتر در جست‌وجو و فیلتر

\subsection{ADR-03: چرا PostgreSQL؟}

\textbf{تصمیم:} PostgreSQL برای Database

\textbf{دلیل:}
- رابطه‌های استاد، درس، واحد و جزوه رابطه‌ای هستند
- قیدهای یکتایی (یک امتیاز برای هر کاربر) به‌سادگی اعمال می‌شود
- رایگان و اوپن‌سورس

\subsection{ADR-04: ذخیرهٔ فایل جزوه‌ها}

\textbf{تصمیم:} فایل‌ها روی دیسک سرور ذخیره می‌شوند و مسیر آن‌ها در PostgreSQL می‌ماند. دسترسی از طریق یک لایهٔ جداگانه (Storage Service) است.

\textbf{دلیل:}
- ساده‌ترین راه برای MVP
- با جداسازی لایه، گذر به سرویس ابری بدون تغییر در بقیهٔ کد ممکن است
- محدودیت: فضای دیسک (حداکثر ۲۰ مگابایت برای هر فایل)

\section{امنیت}

\subsection{احراز هویت}

\begin{itemize}
\tightlist
\item
  \textbf{JWT} برای API
\item
  \textbf{bcrypt} برای رمزنگاری گذرواژه (حداقل ۸ کاراکتر)
\item
  \textbf{Refresh Token} برای تمدید
\end{itemize}

\subsection{مجوزدهی}

\begin{itemize}
\tightlist
\item
  \textbf{Role-Based Access Control (RBAC)}
\item
  نقش‌ها: Student، Professor، Admin (و مهمان بدون توکن برای مشاهده)
\item
  استاد فقط به جزوه‌های مربوط به درس خودش دسترسی دارد
\item
  حساب استاد پس از تأیید مدیر فعال می‌شود
\end{itemize}

\subsection{امنیت API}

\begin{itemize}
\tightlist
\item
  \textbf{HTTPS} اجباری
\item
  \textbf{Rate Limiting} (۱۰۰ درخواست در دقیقه)
\item
  \textbf{CORS} محدود
\item
  \textbf{Input Validation} با Zod
\end{itemize}

\subsection{امنیت فایل و محتوا}

\begin{itemize}
\tightlist
\item
  فقط PDF، JPG، PNG و DOCX تا ۲۰ مگابایت؛ نوع فایل از محتوای آن بررسی می‌شود، نه فقط پسوند
\item
  نام فایل پاک‌سازی می‌شود و فایل خارج از مسیر عمومی ذخیره می‌شود
\item
  لینک گروه فقط با ساختار \texttt{https} ذخیره می‌شود
\item
  نظرها قبل از نمایش برای جلوگیری از XSS پاک‌سازی می‌شوند
\end{itemize}
