\section{مقدمه}

\subsection{هدف سند}

این سند داستان‌های کاربر (User Stories) پروژهٔ \textbf{کلاس‌یار} را تعریف می‌کند.
هر داستان بر پایهٔ قابلیت‌های سند چشم‌انداز (\texttt{docs/01-vision.md}) و
پرسوناهای \texttt{docs/02-requirements/personas.md} نوشته شده و در شماره‌گذاری Sprintها
با برنامهٔ سه‌مرحله‌ای پروژه هماهنگ است.

\subsection{قالب User Story}

هر User Story به این شکل نوشته می‌شود:

\begin{quote}
\textbf{به‌عنوان} {[}نقش{]}، \textbf{می‌خواهم} {[}کاری{]} \textbf{تا} {[}ارزش{]}.
\end{quote}

\subsection{معیارهای INVEST}

\begin{itemize}
\tightlist
\item
  \textbf{I}ndependent — مستقل
\item
  \textbf{N}egotiable — قابل مذاکره
\item
  \textbf{V}aluable — ارزشمند
\item
  \textbf{E}stimable — قابل تخمین
\item
  \textbf{S}mall — کوچک
\item
  \textbf{T}estable — قابل تست
\end{itemize}

\subsection{اولویت‌بندی MoSCoW}

\begin{itemize}
\tightlist
\item
  \textbf{Must:} بدون این، پروژه بی‌معنی است
\item
  \textbf{Should:} مهم، اما اگر وقت نشد بعداً
\item
  \textbf{Could:} خوب است داشته باشیم
\item
  \textbf{Won\textquotesingle t:} در این نسخه نداریم (اپلیکیشن موبایل، پرداخت آنلاین، گفت‌وگوی زنده)
\end{itemize}

\section{معیار پذیرش}

\subsection{US-01}

\textbf{عنوان:} ثبت‌نام و ورود

\textbf{به‌عنوان:} کاربر جدید (دانشجو یا استاد)

\textbf{می‌خواهم:} ثبت‌نام کنم و وارد شوم

\textbf{تا:} از امکانات مخصوص نقشم استفاده کنم

\textbf{قابلیت:} F-01

\textbf{معیار پذیرش:}
- {[} {]} ثبت‌نام با ایمیل، گذرواژه و انتخاب نقش (دانشجو یا استاد) انجام می‌شود
- {[} {]} گذرواژه با bcrypt ذخیره می‌شود و گذرواژهٔ کمتر از ۸ کاراکتر پذیرفته نمی‌شود
- {[} {]} ایمیل تکراری پذیرفته نمی‌شود و پیام خطای روشن نمایش داده می‌شود
- {[} {]} حساب استاد پس از تأیید مدیر فعال می‌شود

\subsection{US-02}

\textbf{عنوان:} مدیریت اطلاعات پایه

\textbf{به‌عنوان:} مدیر سامانه

\textbf{می‌خواهم:} استان، دانشگاه، واحد، درس و استاد را تعریف و ویرایش کنم و درخواست افزودن استاد را بررسی کنم

\textbf{تا:} فیلترها همیشه فهرست درست و کامل داشته باشند

\textbf{قابلیت:} F-02

\textbf{معیار پذیرش:}
- {[} {]} مدیر هر یک از موارد بالا را اضافه، ویرایش و حذف می‌کند
- {[} {]} درخواست دانشجو برای افزودن استاد در پنل مدیر دیده و تأیید یا رد می‌شود
- {[} {]} حذف مورد دارای جزوه مجاز نیست و پیام خطا می‌دهد

\subsection{US-03}

\textbf{عنوان:} آپلود جزوه

\textbf{به‌عنوان:} دانشجوی مشارکت‌کننده

\textbf{می‌خواهم:} جزوه‌ام را با مشخصات استاد، درس، استان، واحد و ترم آپلود کنم

\textbf{تا:} دیگران بتوانند آن را پیدا کنند و از آن استفاده کنند

\textbf{قابلیت:} F-03

\textbf{معیار پذیرش:}
- {[} {]} انتخاب استاد، درس، استان، واحد و ترم اجباری و توضیح اختیاری است
- {[} {]} فقط PDF، JPG، PNG و DOCX تا ۲۰ مگابایت پذیرفته می‌شود
- {[} {]} پس از آپلود موفق، جزوه در فهرست «جزوه‌های من» دیده می‌شود

\subsection{US-04}

\textbf{عنوان:} فهرست جزوه‌ها

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} فهرست جزوه‌ها را با عنوان، استاد و میانگین امتیاز ببینم

\textbf{تا:} بدون جست‌وجوی پراکنده در گروه‌ها جزوه‌های موجود را مرور کنم

\textbf{قابلیت:} F-04, F-05

\textbf{معیار پذیرش:}
- {[} {]} جدیدترین جزوه‌ها به‌صورت صفحه‌بندی‌شده نمایش داده می‌شود
- {[} {]} هر مورد عنوان، استاد، درس و میانگین امتیاز را نشان می‌دهد
- {[} {]} در صورت نبودن جزوه، پیام «جزوه‌ای یافت نشد» نمایش داده می‌شود

\subsection{US-05}

\textbf{عنوان:} جزئیات و دانلود جزوه

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} جزئیات جزوه را ببینم و آن را دانلود کنم

\textbf{تا:} قبل از مطالعه از مناسب‌بودنش مطمئن شوم

\textbf{قابلیت:} F-05

\textbf{معیار پذیرش:}
- {[} {]} عنوان، استاد، درس، ترم، آپلودکننده، میانگین امتیاز و تعداد دانلود نمایش داده می‌شود
- {[} {]} دکمهٔ دانلود فایل اصلی را بدون خطا می‌دهد
- {[} {]} تعداد دانلود پس از هر دانلود یک واحد زیاد می‌شود

\subsection{US-06}

\textbf{عنوان:} مدیریت جزوه‌های من

\textbf{به‌عنوان:} دانشجوی مشارکت‌کننده

\textbf{می‌خواهم:} جزوه‌های آپلودشدهٔ خودم را ویرایش یا حذف کنم

\textbf{تا:} اطلاعات اشتباه را اصلاح کنم و محتوای قدیمی را کنار بگذارم

\textbf{قابلیت:} F-03

\textbf{معیار پذیرش:}
- {[} {]} هر کاربر فقط جزوه‌های خودش را ویرایش یا حذف می‌کند
- {[} {]} پیش از حذف، تأیید کاربر گرفته می‌شود
- {[} {]} تغییر مشخصات بلافاصله در فهرست و جزئیات دیده می‌شود

\subsection{US-07}

\textbf{عنوان:} جست‌وجو و فیلتر

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} جزوه‌ها را بر پایهٔ استاد، استان، واحد، درس و ترم فیلتر کنم

\textbf{تا:} سریع جزوهٔ موردنظرم را پیدا کنم

\textbf{قابلیت:} F-04

\textbf{معیار پذیرش:}
- {[} {]} ترکیب چند فیلتر (مثلاً استاد + استان + واحد) درست کار می‌کند
- {[} {]} نتیجه کمتر از ۲ ثانیه نمایش داده می‌شود
- {[} {]} پاک‌کردن فیلترها فهرست کامل را برمی‌گرداند

\subsection{US-08}

\textbf{عنوان:} امتیاز به جزوه

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} به جزوه امتیاز ۱ تا ۵ بدهم

\textbf{تا:} دیگران کیفیت جزوه را بدانند و بهترین‌ها دیده شوند

\textbf{قابلیت:} F-06

\textbf{معیار پذیرش:}
- {[} {]} امتیاز بین ۱ تا ۵ است و هر کاربر برای هر جزوه فقط یک بار امتیاز می‌دهد
- {[} {]} کاربر می‌تواند امتیاز خودش را ویرایش کند
- {[} {]} میانگین امتیاز بلافاصله به‌روز می‌شود

\subsection{US-09}

\textbf{عنوان:} نظر و پاسخ

\textbf{به‌عنوان:} دانشجو یا استاد

\textbf{می‌خواهم:} روی جزوه و صفحهٔ استاد نظر بنویسم و به نظرها پاسخ بدهم

\textbf{تا:} بازخورد و گفت‌وگوی مفید دربارهٔ محتوا شکل بگیرد

\textbf{قابلیت:} F-09

\textbf{معیار پذیرش:}
- {[} {]} نظر خالی ثبت نمی‌شود
- {[} {]} هر کاربر می‌تواند زیر نظر دیگران پاسخ بدهد
- {[} {]} استاد زیر نظرهای صفحهٔ خودش پاسخ می‌دهد
- {[} {]} کاربر نظر خودش را حذف می‌کند

\subsection{US-10}

\textbf{عنوان:} صفحهٔ استاد

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} صفحهٔ هر استاد را با جزوه‌ها و میانگین امتیازش ببینم

\textbf{تا:} قبل از انتخاب واحد او را بهتر بشناسم

\textbf{قابلیت:} F-07

\textbf{معیار پذیرش:}
- {[} {]} صفحهٔ استاد جزوه‌ها، میانگین امتیاز و نظرها را نشان می‌دهد
- {[} {]} جزوه‌های استاد بر پایهٔ درس قابل فیلتر است
- {[} {]} استاد بدون جزوه صفحهٔ خالی با پیام مناسب دارد

\subsection{US-11}

\textbf{عنوان:} امتیاز به استاد

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} به استاد امتیاز بدهم

\textbf{تا:} دانشجویان بعدی تجربهٔ دیگران را بدانند

\textbf{قابلیت:} F-08

\textbf{معیار پذیرش:}
- {[} {]} هر دانشجو برای هر استاد فقط یک امتیاز ثبت می‌کند
- {[} {]} دانشجو می‌تواند امتیاز خودش را ویرایش کند
- {[} {]} میانگین امتیاز استاد بلافاصله به‌روز می‌شود

\subsection{US-12}

\textbf{عنوان:} نمونه‌سؤال و لینک گروه

\textbf{به‌عنوان:} دانشجوی استفاده‌کننده

\textbf{می‌خواهم:} نمونه‌سؤال‌ها و لینک گروه کلاس هر استاد را ثبت و مشاهده کنم

\textbf{تا:} بهتر برای امتحان آماده شوم و به هم‌کلاسی‌ها وصل شوم

\textbf{قابلیت:} F-10

\textbf{معیار پذیرش:}
- {[} {]} نمونه‌سؤال (فایل) یا لینک گروه برای یک استاد و درس ثبت می‌شود
- {[} {]} فقط لینک https معتبر ذخیره می‌شود
- {[} {]} نمونه‌سؤال‌ها و لینک‌ها در صفحهٔ استاد نمایش داده می‌شود

\subsection{US-13}

\textbf{عنوان:} تأیید جزوه و نمرهٔ تشویقی

\textbf{به‌عنوان:} استاد

\textbf{می‌خواهم:} جزوه‌های درسم را تأیید کنم و برای آپلودکننده نمرهٔ تشویقی ثبت کنم

\textbf{تا:} دانشجویان فعال تشویق شوند

\textbf{قابلیت:} F-11

\textbf{معیار پذیرش:}
- {[} {]} استاد فقط جزوه‌های مربوط به خودش را می‌بیند و تأیید می‌کند
- {[} {]} نمرهٔ تشویقی همراه یادداشت کوتاه ثبت می‌شود و آپلودکننده آن را در پروفایلش می‌بیند
- {[} {]} سامانه صریحاً نشان می‌دهد این نمره رسمی نیست و فقط تشویقی است
- {[} {]} جزوهٔ تأییدشده نشان «تأیید استاد» می‌گیرد

\subsection{US-14}

\textbf{عنوان:} گزارش محتوای نامناسب

\textbf{به‌عنوان:} دانشجو

\textbf{می‌خواهم:} جزوه یا نظر نامناسب را گزارش کنم

\textbf{تا:} محتوای ناقض حقوق دیگران یا توهین‌آمیز بررسی شود

\textbf{قابلیت:} F-12

\textbf{معیار پذیرش:}
- {[} {]} هر کاربر جزوه یا نظر را با ذکر دلیل گزارش می‌کند
- {[} {]} یک کاربر یک محتوا را فقط یک بار گزارش می‌دهد
- {[} {]} گزارش با وضعیت «جدید» برای مدیر ثبت می‌شود

\subsection{US-15}

\textbf{عنوان:} پنل مدیر و بررسی گزارش‌ها

\textbf{به‌عنوان:} مدیر سامانه

\textbf{می‌خواهم:} گزارش‌ها را ببینم و محتوای نامناسب را حذف یا گزارش را رد کنم

\textbf{تا:} اعتبار و کیفیت سامانه حفظ شود

\textbf{قابلیت:} F-13

\textbf{معیار پذیرش:}
- {[} {]} فهرست گزارش‌ها در یک صفحه با وضعیت (جدید، بررسی‌شده، ردشده) نمایش داده می‌شود
- {[} {]} مدیر محتوا را حذف یا گزارش را رد می‌کند و وضعیت ثبت می‌شود
- {[} {]} مدیر کاربران را مشاهده و در صورت نیاز مسدود می‌کند
- {[} {]} فقط نقش مدیر به این پنل دسترسی دارد
